const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, DELETE, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; function jsonResponse(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: { "Content-Type": "application/json; charset=utf-8", ...CORS_HEADERS } }); } function sanitizePublicBroadcastMessage(rawMsg) { const text = String(rawMsg || "").trim(); if (!text) return "شكراً لدعمك!"; const cleaned = text .replace(/\s*\[(?:LBY|MDR)-\d{4}\]/gi, "") .replace(/\s*\[رقم:[^\]]*\]/gi, "") .replace(/\s*\([^)]*(?:رقم|أرقام|تحويل|هاتف|09\d{7,}|[0-9]{4,})[^)]*\)/gi, "") .replace(/\s*(?:آخر\s*4\s*أرقام|اخر\s*4\s*ارقام|رقم\s*(?:الداعم|الهاتف|المحوّل|المحول)?)\s*:?\s*[0-9٠-٩]{1,12}/gi, "") .replace(/\s*رقم\s*:\s*[0-9٠-٩]{1,12}/gi, "") .replace(/\s{2,}/g, " ") .trim(); return cleaned || "شكراً لدعمك!"; } async function ensureSchema(db) { await db.exec(` CREATE TABLE IF NOT EXISTS donations ( id INTEGER PRIMARY KEY AUTOINCREMENT, alert_uid TEXT UNIQUE NOT NULL, donor_name TEXT NOT NULL, amount REAL NOT NULL, message TEXT NOT NULL, network TEXT NOT NULL, created_at TEXT NOT NULL, review_status TEXT NOT NULL DEFAULT 'pending', alert_notified INTEGER NOT NULL DEFAULT 0 ); `); } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: CORS_HEADERS }); } const db = env.DB; if (!db) { return jsonResponse({ ok: false, error: "D1 database binding (DB) is missing" }, 500); } await ensureSchema(db); const url = new URL(request.url); const pathname = url.pathname.replace(/\/+$/, "") || "/"; try { if (request.method === "GET" && pathname === "/api/donations") { const { results } = await db .prepare("SELECT * FROM donations ORDER BY id DESC LIMIT 100") .all(); return jsonResponse({ ok: true, items: results || [] }); } if (request.method === "POST" && pathname === "/api/donations") { const body = await request.json(); const alertUid = String(body.alert_uid || body.id || `alert_${Date.now()}`); const donorName = String(body.donor_name || "").trim(); const amount = Number(body.amount); const message = String(body.message || "دعم مباشر للبث!").trim(); const network = body.network === "almadar" ? "almadar" : "libyana"; const createdAt = String(body.created_at || new Date().toISOString()); if (!donorName) { return jsonResponse({ ok: false, error: "اسم الداعم مطلوب" }, 400); } if (!Number.isFinite(amount) || amount <= 0) { return jsonResponse({ ok: false, error: "قيمة الدعم غير صالحة" }, 400); } await db .prepare( `INSERT INTO donations (alert_uid, donor_name, amount, message, network, created_at, review_status, alert_notified) VALUES (?1, ?2, ?3, ?4, ?5, ?6, 'pending', 0)` ) .bind(alertUid, donorName, amount, message, network, createdAt) .run(); const createdRow = await db .prepare("SELECT * FROM donations WHERE alert_uid = ?1") .bind(alertUid) .first(); return jsonResponse({ ok: true, item: createdRow }, 201); } const approveMatch = pathname.match(/^\/api\/donations\/([^/]+)\/approve$/); if (request.method === "POST" && approveMatch) { const alertUid = decodeURIComponent(approveMatch[1]); const target = await db .prepare("SELECT * FROM donations WHERE alert_uid = ?1") .bind(alertUid) .first(); if (!target) { return jsonResponse({ ok: false, error: "الطلب غير موجود" }, 404); } await db .prepare( `UPDATE donations SET review_status = 'approved', alert_notified = 1 WHERE alert_uid = ?1` ) .bind(alertUid) .run(); const updatedItem = await db .prepare("SELECT * FROM donations WHERE alert_uid = ?1") .bind(alertUid) .first(); return jsonResponse({ ok: true, item: updatedItem, broadcast_alert: { alert_uid: updatedItem.alert_uid, donor_name: updatedItem.donor_name, amount: Number(updatedItem.amount), message: sanitizePublicBroadcastMessage(updatedItem.message), network: updatedItem.network, created_at: updatedItem.created_at, review_status: "approved" } }); } const rejectMatch = pathname.match(/^\/api\/donations\/([^/]+)\/reject$/); if (request.method === "POST" && rejectMatch) { const alertUid = decodeURIComponent(rejectMatch[1]); const target = await db .prepare("SELECT * FROM donations WHERE alert_uid = ?1") .bind(alertUid) .first(); if (!target) { return jsonResponse({ ok: false, error: "الطلب غير موجود" }, 404); } await db .prepare( `UPDATE donations SET review_status = 'rejected', alert_notified = 0 WHERE alert_uid = ?1` ) .bind(alertUid) .run(); const updatedItem = await db .prepare("SELECT * FROM donations WHERE alert_uid = ?1") .bind(alertUid) .first(); return jsonResponse({ ok: true, item: updatedItem }); } if (request.method === "GET" && pathname === "/api/alerts/pending-dispatch") { const row = await db .prepare( `SELECT * FROM donations WHERE review_status = 'approved' AND alert_notified = 1 ORDER BY id ASC LIMIT 1` ) .first(); if (!row) { return jsonResponse({ ok: true, alert: null }); } await db .prepare("UPDATE donations SET alert_notified = 0 WHERE alert_uid = ?1") .bind(row.alert_uid) .run(); return jsonResponse({ ok: true, alert: { alert_uid: row.alert_uid, donor_name: row.donor_name, amount: Number(row.amount), message: sanitizePublicBroadcastMessage(row.message), network: row.network, created_at: row.created_at, review_status: "approved" } }); } if (request.method === "GET" && pathname === "/api/alerts/latest") { return jsonResponse({ ok: true, alert: null }); } if (request.method === "POST" && pathname === "/api/donations/clear") { const body = await request.json().catch(() => ({})); } else { await db.prepare("DELETE FROM donations").run(); } return jsonResponse({ ok: true }); } const deleteMatch = pathname.match(/^\/api\/donations\/([^/]+)$/); if (request.method === "DELETE" && deleteMatch) { const alertUid = decodeURIComponent(deleteMatch[1]); await db.prepare("DELETE FROM donations WHERE alert_uid = ?1").bind(alertUid).run(); return jsonResponse({ ok: true }); } return jsonResponse({ ok: false, error: "Not found" }, 404); } catch (err) { return jsonResponse({ ok: false, error: err.message || "Internal error" }, 500); } } };
